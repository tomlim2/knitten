// JavaScript: Copy code to clipboard
function copyCode(button) {
    const pre = button.nextElementSibling;
    const code = pre.querySelector('code');
    navigator.clipboard.writeText(code.textContent);
}

    display: grid;

Usage Notes
  • Main content: max 720px for optimal reading width
  • Sidebar: 320px fixed, sticky scroll (stays visible)
  • Gap: 24px between columns
  • Mobile: Stacks to single column, sidebar on top
  • Use cases: Standards viewer, learnings browser, documentation

ColorChip

Swatch + hex input color picker. Combines a visual color swatch (wrapping a hidden native <input type="color">) with a hex text input. All state syncs internally.

Live Preview (Dark Context)
Standalone
Outline
% 
import { ColorChip } from './color-chip.js';

// Create
const chip = new ColorChip('#5b9bd5', { ariaLabel: 'Brush color' });
container.appendChild(chip.el);

// Callbacks
chip.onStart  = ()    => { saveUndoState(); };    // first input (drag start)
chip.onChange  = (hex) => { model.color = hex; };  // every change
chip.onCommit = ()    => { finalizeUndo(); };      // change done

// Programmatic update
chip.value = '#ff0000';   // syncs swatch + hex
console.log(chip.value);  // '#ff0000'

// Mount via placeholder
// HTML: <span id="my-chip"></span>
document.getElementById('my-chip').replaceWith(chip.el);

// Push right in flex row
chip.el.style.marginLeft = 'auto';
Required CSS
.color-chip-wrap {
    display: flex;
    align-items: center;
    gap: 4px;
}
.gradient-swatch {
    width: 18px;
    height: 18px;
    border-radius: 3px;
    border: 1px solid rgba(255,255,255,0.15);
    cursor: pointer;
    flex-shrink: 0;
    position: relative;
    overflow: hidden;
}
.gradient-swatch input[type="color"] {
    position: absolute;
    inset: -4px;
    width: calc(100% + 8px);
    height: calc(100% + 8px);
    opacity: 0;
    cursor: pointer;
    border: none;
}
.gradient-hex {
    width: 52px;
    background: transparent;
    border: 1px solid transparent;
    border-radius: 3px;
    color: var(--text-dim);
    font-size: 10px;
    font-family: inherit;
    font-variant-numeric: tabular-nums;
    padding: 2px 4px;
    outline: none;
    text-transform: uppercase;
}
.gradient-hex:hover { border-color: var(--border); }
.gradient-hex:focus { border-color: var(--accent-dim); color: var(--text); }
Design Concept
Aspect Decision
Native picker Hidden <input type="color"> overlays the swatch — OS-native color dialog, zero custom UI
Hex input Direct hex editing for precise values; validates on change, reverts on invalid
3 callbacks onStart (undo snapshot), onChange (live update), onCommit (finalize) — maps to drag lifecycle
stopPropagation Click events stop on input/hex — safe to embed in clickable rows (layers, gradient stops)
Mounting Place <span id="..."> in HTML, call .replaceWith(chip.el) in JS — no framework dependency
